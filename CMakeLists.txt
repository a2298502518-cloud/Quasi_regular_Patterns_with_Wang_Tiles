cmake_minimum_required(VERSION 3.25)
project(QrpWang VERSION 0.2.0 LANGUAGES C CXX)
option(QRP_BUILD_DESKTOP "Build the native OpenGL workbench" ON)
option(QRP_BUILD_TESTS "Build current method checks" ON)
include(FetchContent)
find_package(Threads REQUIRED)

# 本机可复用纯第三方源码缓存；不依赖删除区或旧工程构建产物。
foreach(dep IN ITEMS lodepng nlohmann_json glfw imgui)
    string(TOUPPER "${dep}" dep_upper)
    if(EXISTS "${PROJECT_SOURCE_DIR}/.codex-temp/native-deps/${dep}-src")
        set(FETCHCONTENT_SOURCE_DIR_${dep_upper} "${PROJECT_SOURCE_DIR}/.codex-temp/native-deps/${dep}-src" CACHE PATH "Local dependency source")
    endif()
endforeach()
FetchContent_Declare(lodepng GIT_REPOSITORY https://github.com/lvandeve/lodepng.git GIT_TAG ed6fe5825c6a4fbb7f58ab35a4231c7543cd452a)
FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(lodepng nlohmann_json)
add_library(qrp_png_codec STATIC "${lodepng_SOURCE_DIR}/lodepng.cpp")
target_include_directories(qrp_png_codec SYSTEM PUBLIC "${lodepng_SOURCE_DIR}")

function(qrp_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
add_library(qrp_core STATIC src/qrp/Field.cpp)
target_include_directories(qrp_core PUBLIC src)
target_compile_features(qrp_core PUBLIC cxx_std_20)
qrp_warnings(qrp_core)
add_library(qrp_raster STATIC src/render/Raster.cpp)
target_link_libraries(qrp_raster PUBLIC qrp_core nlohmann_json::nlohmann_json PRIVATE qrp_png_codec Threads::Threads)
qrp_warnings(qrp_raster)
add_executable(qrp_generate apps/export_main.cpp)
target_link_libraries(qrp_generate PRIVATE qrp_raster)
target_compile_definitions(qrp_generate PRIVATE QRP_PROJECT_DIR="${PROJECT_SOURCE_DIR}")
qrp_warnings(qrp_generate)

if(QRP_BUILD_DESKTOP)
    find_package(OpenGL REQUIRED)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 7b6aead9fb88b3623e3b3725ebb42670cbe4c579)
    FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 01380c579715e62fb9a8d6ec0502c4ea83bfde6e)
    FetchContent_MakeAvailable(glfw imgui)
    add_library(qrp_ui STATIC "${imgui_SOURCE_DIR}/imgui.cpp" "${imgui_SOURCE_DIR}/imgui_draw.cpp" "${imgui_SOURCE_DIR}/imgui_tables.cpp" "${imgui_SOURCE_DIR}/imgui_widgets.cpp" "${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp" "${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp")
    target_include_directories(qrp_ui SYSTEM PUBLIC "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends")
    target_link_libraries(qrp_ui PUBLIC glfw OpenGL::GL)
    add_executable(qrp_workbench apps/desktop/main.cpp)
    target_link_libraries(qrp_workbench PRIVATE qrp_raster qrp_ui qrp_png_codec)
    target_compile_definitions(qrp_workbench PRIVATE QRP_PROJECT_DIR="${PROJECT_SOURCE_DIR}")
    qrp_warnings(qrp_workbench)
endif()
if(QRP_BUILD_TESTS)
    enable_testing()
    add_executable(qrp_core_tests tests/native_core_tests.cpp)
    target_link_libraries(qrp_core_tests PRIVATE qrp_core)
    qrp_warnings(qrp_core_tests)
    add_test(NAME current_native_core COMMAND qrp_core_tests)
endif()
